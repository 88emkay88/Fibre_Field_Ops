Who has access: "Anyone" * 9. Click Deploy and copy the Web App URL into APP_SCRIPT_URL in index.html */ // Global Configuration const FOLDER_NAME = "FibreGems_Uploads"; const TIMEZONE = "Africa/Johannesburg"; // SAST (UTC+2) /** * Handle HTTP GET Requests */ function doGet(e) { try { const action = e && e.parameter ? e.parameter.action : null; const ss = SpreadsheetApp.getActiveSpreadsheet(); if (action === "getSuperAdminData" || action === "getPortalData") { return getSuperAdminData(ss); } return createJsonResponse({ status: "success", message: "FibreGems Field Operations API is active." }); } catch (err) { return createJsonResponse({ status: "error", message: err.toString() }); } } /** * Handle HTTP POST Requests */ function doPost(e) { try { if (!e || !e.postData || !e.postData.contents) { return createJsonResponse({ status: "error", message: "No post data received" }); } const data = JSON.parse(e.postData.contents); const action = data.action; const payload = data.payload || {}; const ss = SpreadsheetApp.getActiveSpreadsheet(); switch (action) { case "submitAgentSignOn": case "submitAgentClockIn": return handleAgentSignOn(ss, payload); case "submitLeaderCheckIn": return handleLeaderCheckIn(ss, payload); case "setWeeklyObjective": return handleSetWeeklyObjective(ss, payload); case "logAgentStats": return handleLogAgentStats(ss, payload); default: return createJsonResponse({ status: "error", message: "Invalid action specified: " + action }); } } catch (err) { return createJsonResponse({ status: "error", message: err.toString() }); } } /** * Helper: Create JSON Output with CORS */ function createJsonResponse(data) { return ContentService.createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } /** * Get or create a Google Drive Folder for image uploads */ function getUploadFolder() { const folders = DriveApp.getFoldersByName(FOLDER_NAME); if (folders.hasNext()) { return folders.next(); } const folder = DriveApp.createFolder(FOLDER_NAME); folder.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); return folder; } /** * Save Base64 Image to Google Drive and return its viewable thumbnail/direct URL */ function saveBase64Image(base64Data, filenamePrefix) { if (!base64Data || !base64Data.startsWith("data:image")) { return ""; } try { const parts = base64Data.split(","); const meta = parts[0]; const raw = parts[1]; const mimeMatch = meta.match(/data:(.*?);base64/); const contentType = mimeMatch ? mimeMatch[1] : "image/jpeg"; const ext = contentType.includes("png") ? "png" : "jpg"; const decoded = Utilities.base64Decode(raw); const blob = Utilities.newBlob(decoded, contentType, `${filenamePrefix}_${new Date().getTime()}.${ext}`); const folder = getUploadFolder(); const file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); // Return direct drive file link return file.getUrl(); } catch (err) { Logger.log("Error saving base64 image: " + err); return ""; } } /** * Get or create Sheet Tab with header columns */ function getOrCreateSheet(ss, sheetName, headers) { let sheet = ss.getSheetByName(sheetName); if (!sheet) { sheet = ss.insertSheet(sheetName); if (headers && headers.length > 0) { sheet.appendRow(headers); sheet.getRange(1, 1, 1, headers.length).setFontWeight("bold").setBackground("#e2e8f0"); sheet.setFrozenRows(1); } } return sheet; } /** * 1. Agent Sign-On / Clock-In Handler */ function handleAgentSignOn(ss, payload) { const sheet = getOrCreateSheet(ss, "AgentSignOns", [ "Timestamp", "Date", "Time", "Agent Name", "Team Leader", "Location", "Is Late", "Photo URL", "GPS", "Notes" ]); const now = new Date(); const dateStr = Utilities.formatDate(now, TIMEZONE, "yyyy-MM-dd"); const timeStr = Utilities.formatDate(now, TIMEZONE, "HH:mm:ss"); // Determine if late (after 10:00 AM) const hours = parseInt(Utilities.formatDate(now, TIMEZONE, "HH"), 10); const minutes = parseInt(Utilities.formatDate(now, TIMEZONE, "mm"), 10); const isLate = (hours > 10 || (hours === 10 && minutes > 0)) ? "Yes" : "No"; let photoUrl = ""; if (payload.photoBase64) { photoUrl = saveBase64Image(payload.photoBase64, "Agent_" + (payload.agentName || "Selfie").replace(/\s+/g, "_")); } sheet.appendRow([ now, dateStr, timeStr, payload.agentName || "", payload.leaderName || "", payload.location || "", isLate, photoUrl, payload.gps || "", payload.notes || "" ]); clearCache(); return createJsonResponse({ status: "success", message: "Agent sign-on recorded successfully.", isLate: isLate, photoUrl: photoUrl }); } /** * 2. Team Leader Check-In Handler */ function handleLeaderCheckIn(ss, payload) { const sheet = getOrCreateSheet(ss, "LeaderCheckIns", [ "Timestamp", "Date", "Time", "Leader Name", "Region", "GPS", "Vehicle Photo URL", "Selfie Photo URL" ]); const now = new Date(); const dateStr = Utilities.formatDate(now, TIMEZONE, "yyyy-MM-dd"); const timeStr = Utilities.formatDate(now, TIMEZONE, "HH:mm:ss"); let vehiclePhotoUrl = ""; if (payload.vehiclePhotoBase64) { vehiclePhotoUrl = saveBase64Image(payload.vehiclePhotoBase64, "Leader_Vehicle_" + (payload.leaderName || "Leader").replace(/\s+/g, "_")); } let selfiePhotoUrl = ""; if (payload.selfiePhotoBase64) { selfiePhotoUrl = saveBase64Image(payload.selfiePhotoBase64, "Leader_Selfie_" + (payload.leaderName || "Leader").replace(/\s+/g, "_")); } sheet.appendRow([ now, dateStr, timeStr, payload.leaderName || "", payload.region || "", payload.gps || "", vehiclePhotoUrl, selfiePhotoUrl ]); clearCache(); return createJsonResponse({ status: "success", message: "Leader check-in recorded successfully.", vehiclePhotoUrl: vehiclePhotoUrl, selfiePhotoUrl: selfiePhotoUrl }); } /** * 3. Weekly / Daily Objectives Handler */ function handleSetWeeklyObjective(ss, payload) { const sheet = getOrCreateSheet(ss, "Objectives", [ "Timestamp", "View Type", "Target Date", "Leader Name", "Assigned Location", "Focus Areas", "Target Metrics" ]); const now = new Date(); sheet.appendRow([ now, payload.viewType || "Weekly", payload.targetDate || Utilities.formatDate(now, TIMEZONE, "yyyy-MM-dd"), payload.leaderName || "", payload.assignedLocation || "", payload.focusAreas || "", payload.targetMetrics || "Active Sales" ]); clearCache(); return createJsonResponse({ status: "success", message: "Objective published successfully." }); } /** * 4. Agent Management / Stats Update Handler */ function handleLogAgentStats(ss, payload) { const sheet = getOrCreateSheet(ss, "AgentPerformance", [ "Timestamp", "Agent Name", "Strikes", "Notes" ]); const now = new Date(); sheet.appendRow([ now, payload.agentName || "", payload.strikes || 0, payload.notes || "" ]); return createJsonResponse({ status: "success", message: "Agent performance updated." }); } function clearCache() { try { CacheService.getScriptCache().remove("super_admin_data"); } catch (e) { Logger.log("Cache clear error: " + e); } } /** * 5. Get Super Admin / Portal Data Handler */ function getSuperAdminData(ss) { const cache = CacheService.getScriptCache(); const cached = cache.get("super_admin_data"); if (cached) { return ContentService.createTextOutput(cached).setMimeType(ContentService.MimeType.JSON); } const agentSheet = ss.getSheetByName("AgentSignOns"); const leaderSheet = ss.getSheetByName("LeaderCheckIns"); const agents = []; if (agentSheet && agentSheet.getLastRow() > 1) { const lastRow = agentSheet.getLastRow(); const lastCol = Math.max(agentSheet.getLastColumn(), 10); const data = agentSheet.getRange(1, 1, lastRow, lastCol).getValues(); const headers = data[0]; // Find column indexes const idxDate = headers.indexOf("Date"); const idxTime = headers.indexOf("Time"); const idxName = headers.indexOf("Agent Name"); const idxLeader = headers.indexOf("Team Leader"); const idxLoc = headers.indexOf("Location"); const idxLate = headers.indexOf("Is Late"); const idxPhoto = headers.indexOf("Photo URL"); const idxGPS = headers.indexOf("GPS"); const idxNotes = headers.indexOf("Notes"); for (let i = 1; i < data.length; i++) { const row = data[i]; if (!row[idxName] && !row[0]) continue; agents.push({ date: idxDate >= 0 ? row[idxDate] : row[1], time: idxTime >= 0 ? row[idxTime] : row[2], agentName: idxName >= 0 ? row[idxName] : row[3], leaderName: idxLeader >= 0 ? row[idxLeader] : row[4], location: idxLoc >= 0 ? row[idxLoc] : row[5], isLate: idxLate >= 0 ? row[idxLate] : row[6], photoUrl: idxPhoto >= 0 ? row[idxPhoto] : row[7], gps: idxGPS >= 0 ? row[idxGPS] : row[8], notes: idxNotes >= 0 ? row[idxNotes] : row[9] }); } } const leaders = []; if (leaderSheet && leaderSheet.getLastRow() > 1) { const lastRow = leaderSheet.getLastRow(); const lastCol = Math.max(leaderSheet.getLastColumn(), 8); const data = leaderSheet.getRange(1, 1, lastRow, lastCol).getValues(); const headers = data[0]; const idxDate = headers.indexOf("Date"); const idxTime = headers.indexOf("Time"); const idxName = headers.indexOf("Leader Name"); const idxRegion = headers.indexOf("Region"); const idxGPS = headers.indexOf("GPS"); const idxVeh = headers.indexOf("Vehicle Photo URL"); const idxPhoto = headers.indexOf("Selfie Photo URL"); for (let i = 1; i < data.length; i++) { const row = data[i]; if (!row[idxName] && !row[0]) continue; leaders.push({ date: idxDate >= 0 ? row[idxDate] : row[1], time: idxTime >= 0 ? row[idxTime] : row[2], leaderName: idxName >= 0 ? row[idxName] : row[3], region: idxRegion >= 0 ? row[idxRegion] : row[4], gps: idxGPS >= 0 ? row[idxGPS] : row[5], vehicleInfo: idxVeh >= 0 ? row[idxVeh] : row[6], photoUrl: idxPhoto >= 0 ? row[idxPhoto] : row[7] }); } } const outputPayload = { status: "success", agents: agents.reverse(), // most recent first leaders: leaders.reverse() // most recent first }; const jsonStr = JSON.stringify(outputPayload); if (jsonStr.length < 95000) { try { cache.put("super_admin_data", jsonStr, 180); // cache for 3 minutes } catch (e) {} } return ContentService.createTextOutput(jsonStr).setMimeType(ContentService.MimeType.JSON); }
